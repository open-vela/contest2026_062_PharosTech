cmake_minimum_required(VERSION 3.16)
project(nyamp_vision LANGUAGES CXX)
add_library(nyamp_vision STATIC nyamp_vision.cpp nyamp_identity_math.cpp nyamp_tracker.cpp)
target_include_directories(nyamp_vision PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(nyamp_vision PUBLIC cxx_std_17)
target_compile_options(nyamp_vision PRIVATE -Wall -Wextra -Werror)
add_executable(nyamp_vision_test nyamp_vision_test.cpp)
target_link_libraries(nyamp_vision_test PRIVATE nyamp_vision)
target_compile_options(nyamp_vision_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
enable_testing()
add_test(NAME nyamp_vision_test COMMAND nyamp_vision_test)
set(NYAMP_RKNN_INCLUDE "" CACHE PATH "External RKNN headers")
set(NYAMP_RKNN_LIBRARY "" CACHE FILEPATH "External RKNN runtime")
if(NYAMP_RKNN_INCLUDE AND NYAMP_RKNN_LIBRARY)
 add_library(nyamp_vision_rknn STATIC nyamp_vision_rknn.cpp nyamp_tracker_rknn.cpp)
 target_include_directories(nyamp_vision_rknn PRIVATE ${NYAMP_RKNN_INCLUDE})
 target_link_libraries(nyamp_vision_rknn PUBLIC nyamp_vision ${NYAMP_RKNN_LIBRARY})
 target_compile_options(nyamp_vision_rknn PRIVATE -Wall -Wextra -Werror)
endif()

add_executable(nyamp_tracker_test nyamp_tracker_test.cpp)
target_link_libraries(nyamp_tracker_test PRIVATE nyamp_vision)
target_compile_options(nyamp_tracker_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME nyamp_tracker_test COMMAND nyamp_tracker_test)
