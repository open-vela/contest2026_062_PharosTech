cmake_minimum_required(VERSION 3.16)
project(nyamp_models LANGUAGES CXX)

find_package(Threads REQUIRED)
add_library(nyamp_models STATIC nyamp_models.cpp)
target_include_directories(nyamp_models PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(nyamp_models PUBLIC cxx_std_17)
target_compile_options(nyamp_models PRIVATE -Wall -Wextra -Werror)
target_link_libraries(nyamp_models PUBLIC Threads::Threads)

enable_testing()
add_executable(nyamp_models_replay_test nyamp_models_replay_test.cpp)
target_compile_options(nyamp_models_replay_test PRIVATE -Wall -Wextra -Werror)
target_link_libraries(nyamp_models_replay_test PRIVATE nyamp_models)
add_test(NAME nyamp_models_replay_test COMMAND nyamp_models_replay_test)

set(NYAMP_SHERPA_INCLUDE "" CACHE PATH "Directory containing external c-api.h")
set(NYAMP_SHERPA_LIBRARY "" CACHE FILEPATH "External libsherpa-onnx-c-api.so")
if(NYAMP_SHERPA_INCLUDE AND NYAMP_SHERPA_LIBRARY)
  add_library(nyamp_sherpa STATIC nyamp_sherpa.cpp)
  target_include_directories(nyamp_sherpa PRIVATE ${NYAMP_SHERPA_INCLUDE})
  target_link_libraries(nyamp_sherpa PUBLIC nyamp_models ${NYAMP_SHERPA_LIBRARY})
  target_compile_options(nyamp_sherpa PRIVATE -Wall -Wextra -Werror)
  add_executable(nyamp_asr_file_test nyamp_asr_file_test.cpp)
  target_include_directories(nyamp_asr_file_test PRIVATE ${NYAMP_SHERPA_INCLUDE})
  target_link_libraries(nyamp_asr_file_test PRIVATE nyamp_sherpa)
  target_compile_options(nyamp_asr_file_test PRIVATE -Wall -Wextra -Werror)
endif()
