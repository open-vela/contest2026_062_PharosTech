cmake_minimum_required(VERSION 3.16)
project(nyamp_models LANGUAGES CXX)

find_package(Threads REQUIRED)
add_library(nyamp_models STATIC nyamp_models.cpp)
target_include_directories(nyamp_models PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(nyamp_models PUBLIC cxx_std_17)
target_compile_options(nyamp_models PRIVATE -Wall -Wextra -Werror)
target_link_libraries(nyamp_models PUBLIC Threads::Threads)

enable_testing()
add_executable(nyamp_models_replay_test nyamp_models_replay_test.cpp)
target_compile_options(nyamp_models_replay_test PRIVATE -Wall -Wextra -Werror)
target_link_libraries(nyamp_models_replay_test PRIVATE nyamp_models)
add_test(NAME nyamp_models_replay_test COMMAND nyamp_models_replay_test)

set(NYAMP_SHERPA_INCLUDE "" CACHE PATH "Directory containing external c-api.h")
set(NYAMP_SHERPA_LIBRARY "" CACHE FILEPATH "External libsherpa-onnx-c-api.so")
if(NYAMP_SHERPA_INCLUDE AND NYAMP_SHERPA_LIBRARY)
  add_library(nyamp_sherpa STATIC nyamp_sherpa.cpp)
  target_include_directories(nyamp_sherpa PRIVATE ${NYAMP_SHERPA_INCLUDE})
  target_link_libraries(nyamp_sherpa PUBLIC nyamp_models ${NYAMP_SHERPA_LIBRARY})
  target_compile_options(nyamp_sherpa PRIVATE -Wall -Wextra -Werror)
  add_executable(nyamp_asr_file_test nyamp_asr_file_test.cpp)
  target_include_directories(nyamp_asr_file_test PRIVATE ${NYAMP_SHERPA_INCLUDE})
  target_link_libraries(nyamp_asr_file_test PRIVATE nyamp_sherpa)
  target_compile_options(nyamp_asr_file_test PRIVATE -Wall -Wextra -Werror)
endif()

set(NYAMP_ORT_INCLUDE "" CACHE PATH "Directory containing external onnxruntime_c_api.h")
set(NYAMP_ORT_LIBRARY "" CACHE FILEPATH "External libonnxruntime.so")
set(NYAMP_RKNN_INCLUDE "" CACHE PATH "Directory containing external rknn_api.h")
set(NYAMP_RKNN_LIBRARY "" CACHE FILEPATH "External librknnrt.so")
if(NYAMP_ORT_INCLUDE AND NYAMP_ORT_LIBRARY AND NYAMP_RKNN_INCLUDE AND NYAMP_RKNN_LIBRARY)
  add_library(nyamp_melo STATIC nyamp_melo.cpp)
  target_include_directories(nyamp_melo PRIVATE ${NYAMP_ORT_INCLUDE} ${NYAMP_RKNN_INCLUDE})
  target_link_libraries(nyamp_melo PUBLIC nyamp_models ${NYAMP_ORT_LIBRARY} ${NYAMP_RKNN_LIBRARY})
  target_compile_options(nyamp_melo PRIVATE -Wall -Wextra -Werror)
  add_executable(nyamp_tts_file nyamp_model_file.cpp)
  target_link_libraries(nyamp_tts_file PRIVATE nyamp_melo)
  target_compile_options(nyamp_tts_file PRIVATE -Wall -Wextra -Werror)
endif()

set(NYAMP_RKLLM_ROOT "" CACHE PATH "Directory containing external rkllm.h and librkllmrt.so")
if(NYAMP_RKLLM_ROOT)
  add_library(nyamp_rkllm STATIC nyamp_rkllm.cpp)
  target_include_directories(nyamp_rkllm PRIVATE ${NYAMP_RKLLM_ROOT})
  target_link_libraries(nyamp_rkllm PUBLIC nyamp_models ${NYAMP_RKLLM_ROOT}/librkllmrt.so)
  target_compile_options(nyamp_rkllm PRIVATE -Wall -Wextra -Werror)
  add_executable(nyamp_llm_file nyamp_model_file.cpp)
  target_compile_definitions(nyamp_llm_file PRIVATE NYAMP_FILE_LLM=1)
  target_link_libraries(nyamp_llm_file PRIVATE nyamp_rkllm)
  target_compile_options(nyamp_llm_file PRIVATE -Wall -Wextra -Werror)
endif()
